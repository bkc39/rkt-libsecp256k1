
;;;; tweaking a secret key and its public key agree

> child
#<secret-key>
> (secret-key->public-key child)
#<public-key 032699aa9999676d80f83f2fd1697e1c0673272d45396500c7d7719c8fe5bd489f>
> (equal? (secret-key->public-key child) (public-key-tweak-add g tweak))
#t

;;;; a zero tweak changes nothing

> (equal? (public-key-tweak-add g zero) g)
#t
> (equal? (secret-key->public-key (secret-key-tweak-add one zero)) g)
#t

;;;; negation

> (public-key-negate g)
#<public-key 0379be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798>
> (equal? (secret-key->public-key (secret-key-negate one)) (public-key-negate g))
#t
> (equal? (public-key-negate (public-key-negate g)) g)
#t

;;;; sums of public keys

> (bytes->hex-string (public-key->bytes (public-key-combine g g)))
"02c6047f9441ed7d6d3045406e95c07cd85c778e4b8cef3ca7abac09b95c709ee5"
> (equal? (public-key-combine g g) (secret-key->public-key two))
#t
> (equal?
   (public-key-combine g (secret-key->public-key two))
   (public-key-tweak-add g (secret-key->bytes two)))
#t

;;;; sums that are the point at infinity

> (public-key-combine g (public-key-negate g))
; exn:fail:libsecp256k1:key
; public-key-combine: the sum is the point at infinity
> (secret-key-tweak-add one (secret-key->bytes (secret-key-negate one)))
; exn:fail:libsecp256k1:key
; secret-key-tweak-add: invalid tweak;
;  the tweak is not less than the curve order, or the sum is zero
> (public-key-tweak-add g (make-bytes 32 255))
; exn:fail:libsecp256k1:key
; public-key-tweak-add: invalid tweak;
;  the tweak is not less than the curve order, or the sum is the point at infinity
